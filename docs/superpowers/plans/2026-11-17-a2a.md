# A2A Adapter Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: `superpowers:executing-plans` (or `subagent-driven-development`). Steps use checkbox (`- [ ]`) syntax.

**Goal:** Expose designated agents (with an agent card) over A2A v1.0 at `/a2a/{agent}`, authenticated by API keys, driving the harness.

**Architecture:** `a2a-sdk` 1.3 Starlette routes (`create_agent_card_routes` + `create_jsonrpc_routes` + `DefaultRequestHandler`) behind a per-agent dispatcher; an `AgentExecutor` maps an A2A context to a harness session (Project at `A2A_WORKSPACE`, `SessionRow`, `SQLiteMessageLogStore`) and streams text as artifacts.

**Tech Stack:** `a2a-sdk` 1.3 (protobuf types), FastAPI/Starlette, pydantic-ai, SQLModel.

## Global Constraints

- Use the official SDK for all A2A wire I/O.
- `/a2a/*` requires a valid API key with scope `sessions:create`.
- Only agents with a stored card are exposed (404 otherwise).
- `A2A_WORKSPACE` (default `.`) is the project root; `A2A_PUBLIC_URL` (default `http://localhost:8000`) fills card URLs.
- `ruff`/`ty`/`bandit` clean; existing tests keep passing.

```python
            s.add(row)

            row.card = json.dumps(body)
            s.add(row)
```

- [ ] **Step 1: Implement `a2a_server.py` (dispatcher + card build)**

```python
import contextvars
import json

from a2a import types as a2a_types
from a2a.server.request_handlers import DefaultRequestHandler
from a2a.server.routes import create_agent_card_routes, create_jsonrpc_routes
from a2a.server.tasks import InMemoryTaskStore
from google.protobuf import json_format
from starlette.applications import Starlette
from starlette.responses import JSONResponse, PlainTextResponse
from sqlmodel import select

from agentbox_models.tables.core import Agent, AgentCard

a2a_user: contextvars.ContextVar[str | None] = contextvars.ContextVar("a2a_user", default=None)


def build_card(stored: dict, agent_name: str, public_url: str):
    card = json_format.ParseDict(stored, a2a_types.AgentCard())
    card.name = stored.get("name") or agent_name
    card.capabilities.streaming = True
    card.supported_interfaces.clear()
    card.supported_interfaces.append(
        a2a_types.AgentInterface(
            protocol_binding="JSONRPC",
            url=f"{public_url}/a2a/{agent_name}",
            protocol_version="1.0",
        )
    )
    return card


class A2ADispatcher:
    def __init__(self, session_factory, settings):
        self._factory = session_factory
        self._settings = settings
        self._apps: dict[str, Starlette] = {}

    async def _agent_with_card(self, name: str):
        async with self._factory() as s:
            agent = (await s.execute(select(Agent).where(Agent.name == name))).scalars().first()
            if agent is None:
                return None, None
            row = (await s.execute(select(AgentCard).where(AgentCard.agent_id == agent.id))).scalars().first()
        return agent, row

    async def _app_for(self, name: str, stored: dict) -> Starlette:
        app = self._apps.get(name)
        if app is not None:
            return app
        card = build_card(stored, name, self._settings.a2a_public_url)
        from agentbox_api.a2a_executor import AgentboxA2AExecutor

        handler = DefaultRequestHandler(
            agent_executor=AgentboxA2AExecutor(name, self._factory, self._settings),
            task_store=InMemoryTaskStore(),
            agent_card=card,
        )
        routes = [*create_agent_card_routes(card), *create_jsonrpc_routes(handler, "/")]
        app = Starlette(routes=routes)
        self._apps[name] = app
        return app

    async def __call__(self, scope, receive, send):
        if scope["type"] != "http":
            return
        path = scope["path"].removeprefix("/a2a").lstrip("/")
        name, _, rest = path.partition("/")
        if not name:
            return await PlainTextResponse("Not found", status_code=404)(scope, receive, send)
        user = scope.get("user")
        auth = scope.get("auth")
        if user is None or not getattr(user, "is_authenticated", False):
            return await JSONResponse({"error": "unauthorized"}, status_code=401)(scope, receive, send)
        if auth is None or "sessions:create" not in auth.scopes:
            return await JSONResponse({"error": "forbidden"}, status_code=403)(scope, receive, send)
        _, row = await self._agent_with_card(name)
        if row is None:
            return await PlainTextResponse("Unknown agent", status_code=404)(scope, receive, send)
        app = await self._app_for(name, json.loads(row.card))
        token = a2a_user.set(getattr(user, "identity", None))
        try:
            rest_scope = dict(scope)
            rest_scope["path"] = "/" + rest if rest else "/"
            rest_scope["raw_path"] = rest_scope["path"].encode()
            await app(rest_scope, receive, send)
        finally:
            a2a_user.reset(token)
```

- [ ] **Step 1: Implement `a2a_executor.py`**

```python
import uuid

from a2a import types as a2a_types
from a2a.helpers import get_message_text, new_task_from_user_message, new_text_message, new_text_part
from a2a.server.agent_execution import AgentExecutor, RequestContext
from a2a.server.events import EventQueue
from a2a.server.tasks import TaskUpdater
from pydantic_ai.messages import PartDeltaEvent, PartStartEvent, TextPart, TextPartDelta
from sqlmodel import select

from agentbox_api.a2a_server import a2a_user
from agentbox_api.runtime import build_registry, resolve_agent_runtime
from agentbox_harness.events import EventBus
from agentbox_harness.harness import Harness
from agentbox_harness.session import Session, Turn
from agentbox_models.repos.message_log import SQLiteMessageLogStore
from agentbox_models.tables.auth import User
from agentbox_models.tables.core import Project, SessionRow, TurnRow


class AgentboxA2AExecutor(AgentExecutor):
    def __init__(self, agent_name, factory, settings):
        self._agent_name = agent_name
        self._factory = factory
        self._settings = settings
        self._runs = {}

    async def _user_id(self) -> str:
        uid = a2a_user.get()
        if uid:
            return uid
        async with self._factory() as s:
            user = (await s.execute(select(User).where(User.username == "admin"))).scalars().first()
        return user.id

    async def execute(self, context: RequestContext, event_queue: EventQueue) -> None:
        task = context.current_task or new_task_from_user_message(context.message)
        if context.current_task is None:
            await event_queue.enqueue_event(task)
        updater = TaskUpdater(event_queue, task.id, task.context_id)
        await updater.start_work()
        session_id = task.context_id
        runtime = await resolve_agent_runtime(self._factory, self._agent_name)
        owner = await self._user_id()
        async with self._factory() as s:
            project = (await s.execute(select(Project).where(Project.path == self._settings.a2a_workspace))).scalars().first()
            if project is None:
                project = Project(id=uuid.uuid4().hex, name=self._settings.a2a_workspace, path=self._settings.a2a_workspace, owner_id=owner)
                s.add(project)
            if (await s.execute(select(SessionRow).where(SessionRow.id == session_id))).scalars().first() is None:
                s.add(SessionRow(id=session_id, project_id=project.id, agent_id=runtime.agent_id))
            await s.commit()
        head = await _latest_leaf(self._factory, session_id)
        store = SQLiteMessageLogStore(self._factory, session_id)
        session = Session(session_id=session_id, project_root=self._settings.a2a_workspace, agent_spec=runtime.spec, store=store, head_id=head, mode_name="agent")
        harness = Harness(registry=build_registry(self._settings.a2a_workspace), model=runtime.model, store_factory=lambda: SQLiteMessageLogStore(self._factory, session_id))
        bus = EventBus(); sub = bus.subscribe()
        run = await harness.run(session, get_message_text(context.message), bus)
        self._runs[session_id] = run

        async def finish():
            await run.await_outcome(); bus.close()

        closer = asyncio.create_task(finish())
        try:
            async for ev in sub.stream():
                chunk = _text_chunk(ev.event)
                if chunk:
                    await updater.add_artifact(parts=[new_text_part(text=chunk)], append=True, last_chunk=False)
        finally:
            closer.cancel()
        outcome = await run.await_outcome()
        self._runs.pop(session_id, None)
        if outcome.status == "success":
            result = await run.await_result()
            await session.append(Turn(id=uuid.uuid4().hex, role="assistant", content=str(result.output), parent_id=session.head_id), parent_id=session.head_id)
            await updater.complete()
        else:
            await updater.failed(message=new_text_message(outcome.message or "error"))

    async def cancel(self, context: RequestContext, event_queue: EventQueue) -> None:
        run = self._runs.get(context.context_id or "")
        if run is not None:
            run.cancel()
        updater = TaskUpdater(event_queue, context.task_id, context.context_id)
        await updater.cancel()


def _text_chunk(event):
    if isinstance(event, PartStartEvent) and isinstance(event.part, TextPart):
        return event.part.content or None
    if isinstance(event, PartDeltaEvent) and isinstance(event.delta, TextPartDelta):
        return event.delta.content_delta or None
    return None


async def _latest_leaf(factory, session_id):
    async with factory() as s:
        turns = (await s.execute(select(TurnRow).where(TurnRow.session_id == session_id))).scalars().all()
    if not turns:
        return None
    parents = {t.parent_id for t in turns if t.parent_id}
    leaves = [t for t in turns if t.id not in parents]
    return leaves[-1].id if leaves else None
```

```ts
export type AgentCard = Record<string, unknown>;
export async function fetchAgentCard(name: string): Promise<AgentCard | null> {
  const { data, response } = await api.GET("/agents/{name}/card", { params: { path: { name } } });
  if (response.status === 404) return null;
  return (data as AgentCard) ?? null;
}
export async function setAgentCard(name: string, card: AgentCard): Promise<AgentCard> {
  const { data, error } = await api.PUT("/agents/{name}/card", { params: { path: { name } }, body: card });
  if (error) throw new Error("Failed to save agent card");
  return data as AgentCard;
}
export async function deleteAgentCard(name: string): Promise<void> {
  await api.DELETE("/agents/{name}/card", { params: { path: { name } } });
}
```
