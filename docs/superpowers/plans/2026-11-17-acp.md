# ACP Adapter Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: `superpowers:executing-plans` (or `subagent-driven-development`). Steps use checkbox (`- [ ]`) syntax.

**Goal:** Run Agentbox as an ACP (v1) agent over stdio so external IDEs can drive sessions, reusing the harness and core session storage.

**Architecture:** `agent-client-protocol` 1.0 (`acp.Agent` + `run_agent`) over stdio. The agent maps ACP sessions to core `Project`/`SessionRow`/`TurnRow` + the harness (`Session`, `EventBus`, `SQLiteMessageLogStore`); prompt blocks map to `str`/`BinaryContent`; stream events map to ACP `session/update`s.

**Tech Stack:** `agent-client-protocol` 1.0, pydantic-ai, FastMCP stdio transport, SQLModel, pytest-asyncio.

## Global Constraints

- Use the official SDK for all wire I/O and schema models (no homegrown JSON-RPC).
- Advertise only implemented capabilities; never emit non-advertised updates.
- `--agent <name>` CLI selects the agent; `cwd` is the project root; `DATABASE_URL` selects the DB.
- Never log MCP `env`/`headers` values.
- `ruff`/`ty`/`bandit` clean; existing tests keep passing.

    box = FilesystemToolbox(pathlib.Path(project_root))
    registry = ToolRegistry()
    registry.register_native("read_text_file", box.read_text_file, required_flags=("read",))

@dataclass
class AgentRuntime:
    agent_id: str
    created_by: str | None
    spec: AgentSpec
    model: object


async def resolve_agent_runtime(factory, agent_name: str) -> AgentRuntime:
    async with factory() as s:
        agent = (
            await s.execute(select(Agent).where(Agent.name == agent_name))
        ).scalars().first()
        if agent is None:
            raise KeyError(agent_name)
        model = (
            await s.execute(select(Model).where(Model.id == agent.model_id))
        ).scalars().first()
        provider = (
            await s.execute(select(Provider).where(Provider.id == model.provider_id))
        ).scalars().first()
    spec = await AgentRepository(factory).load(agent_name)
    return AgentRuntime(
        agent_id=agent.id,
        created_by=agent.created_by,
        spec=spec,
        model=resolve_model(provider, model.name),
    )

- [ ] **Step 2: Implement `prompt_parts`**

```python
import base64

from pydantic_ai.messages import BinaryContent


def prompt_parts(blocks) -> list:
    parts = []
    for b in blocks:
        kind = getattr(b, "type", None)
        if kind == "text":
            parts.append(b.text)
        elif kind == "resource_link":
            mime = f" ({b.mime_type})" if getattr(b, "mime_type", None) else ""
            parts.append(f"[resource: {b.name} {b.uri}{mime}]")
        elif kind == "image":
            parts.append(BinaryContent(data=base64.b64decode(b.data), media_type=b.mime_type))
        elif kind == "audio":
            parts.append(BinaryContent(data=base64.b64decode(b.data), media_type=b.mime_type))
        elif kind == "resource":
            r = b.resource
            text = getattr(r, "text", None)
            blob = getattr(r, "blob", None)
            if text is not None:
                parts.append(f"[resource {r.uri}]\n{text}")
            elif blob is not None:
                parts.append(BinaryContent(data=base64.b64decode(blob), media_type=r.mime_type))
            else:
                raise ValueError("invalid embedded resource")
        else:
            raise ValueError(f"unsupported content block: {kind}")
    return parts
```

```python
from acp import schema as s
from pydantic_ai.messages import (
    FunctionToolCallEvent,
    FunctionToolResultEvent,
    PartDeltaEvent,
    TextPartDelta,
    ThinkingPartDelta,
)


def update_for_event(event):
    if isinstance(event, PartDeltaEvent):
        delta = event.delta
        text = getattr(delta, "content_delta", None)
        if isinstance(delta, TextPartDelta) and text:
            return s.AgentMessageChunk(content=s.TextContentBlock(text=text))
        if isinstance(delta, ThinkingPartDelta) and text:
            return s.AgentThoughtChunk(content=s.TextContentBlock(text=text))
        return None
    if isinstance(event, FunctionToolCallEvent):
        p = event.part
        return s.ToolCallStart(
            tool_call_id=p.tool_call_id, title=p.tool_name, name=p.tool_name,
            kind="other", status="pending",
        )
    if isinstance(event, FunctionToolResultEvent):
        p = event.part
        text = event.content if event.content is not None else str(p.content)
        return s.ToolCallProgress(
            tool_call_id=p.tool_call_id,
            status="completed",
            content=[s.ContentToolCallContent(content=s.TextContentBlock(text=text))],
        )
    return None
```

        sess = self._sessions.get(session_id)
        if sess is None:
            raise acp.RequestError(-32002, "unknown session")
        parts = prompt_parts(prompt)
        text = "".join(p for p in parts if isinstance(p, str))
        harness = Harness(

        hs = Session(

        run = await harness.run(hs, text, bus, prompt_override=parts)

    async def delete_session(self, session_id, **kw):
        async with self._factory() as db:
            row = (await db.execute(select(SessionRow).where(SessionRow.id == session_id))).scalars().first()
            if row is not None:
                for t in (await db.execute(select(TurnRow).where(TurnRow.session_id == session_id))).scalars().all():
                    await db.delete(t)
                for a in (await db.execute(select(SessionAttachment).where(SessionAttachment.session_id == session_id))).scalars().all():
                    await db.delete(a)
                await db.delete(row)
                await db.commit()
        self._sessions.pop(session_id, None)
        return s.DeleteSessionResponse()

    async def close_session(self, session_id, **kw):
        sess = self._sessions.pop(session_id, None)
        if sess is not None and sess.run is not None:
            sess.run.cancel()
        return s.CloseSessionResponse()

    async def _restore(self, session_id, cwd, mcp_servers):
        rt = await resolve_agent_runtime(self._factory, self._agent_name)
        async with self._factory() as db:
            row = (await db.execute(select(SessionRow).where(SessionRow.id == session_id))).scalars().first()
            if row is None or row.agent_id != rt.agent_id:
                raise acp.RequestError(-32002, "unknown session")
            project = (await db.execute(select(Project).where(Project.id == row.project_id))).scalars().first()
        spec = self._with_servers(rt.spec, mcp_servers)
        sess = AcpSession(session_id, project.path, project.id, rt.agent_id, spec, rt.model, _default_mode(spec))
        self._sessions[session_id] = sess
        return sess

    async def _admin_id(self) -> str:
        async with self._factory() as db:
            user = (await db.execute(select(User).where(User.username == "admin"))).scalars().first()
        if user is None:
            raise RuntimeError("no admin user to own ACP projects")
        return user.id

    def _with_servers(self, spec, mcp_servers):
        extra = [
            McpToolBinding(
                id=_slug(server.name) or f"acp-{i}",
                command=server.command,
                args=list(server.args),
                env={e.name: e.value for e in server.env},
            )
            for i, server in enumerate(mcp_servers or [])
        ]
        if not extra:
            return spec
        return spec.model_copy(update={"tool_bindings": [*spec.tool_bindings, *extra]})


def _slug(value: str) -> str:
    import re
    return re.sub(r"[^a-z0-9]+", "-", value.lower()).strip("-")


def _default_mode(spec) -> str:
    names = [m.name for m in spec.mode_definitions] or ["ask", "agent"]
    return "agent" if "agent" in names else names[0]


def _modes(spec, current: str) -> s.SessionModeState:
    modes = spec.mode_definitions or []
    if not modes:
        from agentbox_harness.spec import DEFAULT_MODES
        modes = list(DEFAULT_MODES)
    return s.SessionModeState(
        current_mode_id=current,
        available_modes=[s.SessionMode(id=m.name, name=m.name.capitalize()) for m in modes],
    )


async def _latest_leaf(factory, session_id: str):
    async with factory() as db:
        turns = (await db.execute(select(TurnRow).where(TurnRow.session_id == session_id))).scalars().all()
    if not turns:
        return None
    parents = {t.parent_id for t in turns if t.parent_id}
    leaves = [t for t in turns if t.id not in parents]
    return leaves[-1].id if leaves else None

class CollectingClient(Client):
    def __init__(self):
        self.updates = []

async def test_acp_stdio_end_to_end(tmp_path):
