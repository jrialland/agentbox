# API Keys (lifetime + scope validation + page) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:executing-plans` (or `subagent-driven-development`). Steps use checkbox (`- [ ]`) syntax.

**Goal:** Personal API keys with optional expiry, scope-subset validation, soft revocation, a safe read model, and an `/api-keys` page.

**Architecture:** Extend `routers/api_keys.py` (validate scopes ⊆ caller scopes, accept/normalize `expires_at`, personal `ApiKeyRead` without `key_hash`, soft-revoke). Enforce expiry/revocation in `auth_backend`. Add an `/api-keys` page (list/create with a scopes MultiSelect + expiry preset/one-time key modal/revoke), gated by `api_keys:manage`.

**Tech Stack:** FastAPI, SQLModel, React 19, Mantine v9, TanStack Query, Playwright.

## Global Constraints

- Never return `key_hash`; the raw key is shown only once.
- Scopes must be a subset of the caller's effective scopes (`request.auth.scopes`).
- `expires_at` is optional, UTC, must be in the future at creation.
- Delete = soft revoke (`revoked_at`), row kept.
- Personal: list/revoke only the caller's keys.
- `ruff`/`ty`/`bandit` clean; `bun run test`/`bun run build` pass.

```python
class ApiKeyCreate(BaseModel):
    scopes: list[str] = []
    expires_at: datetime | None = None


class ApiKeyRead(BaseModel):
    id: str
    scopes: list[str]
    expires_at: datetime | None = None
    revoked_at: datetime | None = None
    created_at: datetime


class ApiKeyCreated(ApiKeyRead):
    key: str


def _read(row: ApiKey) -> ApiKeyRead:
    return ApiKeyRead(
        id=row.id,
        scopes=json.loads(row.scopes or "[]"),
        expires_at=row.expires_at,
        revoked_at=row.revoked_at,
        created_at=row.created_at,
    )


def _normalize_expiry(value: datetime | None) -> datetime | None:
    if value is None:
        return None
    if value.tzinfo is None:
        value = value.replace(tzinfo=UTC)
    if value <= datetime.now(UTC):
        raise HTTPException(status_code=400, detail="expires_at must be in the future")
    return value


@router.post("", response_model=ApiKeyCreated)
async def create_api_key(
    body: ApiKeyCreate, request: Request, _=Depends(require_permission("api_keys:manage"))
):
    scopes = list(dict.fromkeys(body.scopes))
    caller = set(request.auth.scopes) if request.auth is not None else set()
    disallowed = [s for s in scopes if s not in caller]
    if disallowed:
        raise HTTPException(
            status_code=400, detail=f"Unauthorized scopes: {', '.join(disallowed)}"
        )
    expires_at = _normalize_expiry(body.expires_at)
    raw = secrets.token_urlsafe(32)
    repo = ApiKeyRepository(request.app.state.session_factory)
    key = ApiKey(
        id=uuid.uuid4().hex,
        user_id=request.user.identity,
        key_hash=hash_api_key(raw),
        scopes=json.dumps(scopes),
        expires_at=expires_at,
    )
    await repo.create(key)
    return ApiKeyCreated(key=raw, **_read(key).model_dump())


@router.get("", response_model=list[ApiKeyRead])
async def list_api_keys(
    request: Request, _=Depends(require_permission("api_keys:manage"))
):
    async with request.app.state.session_factory() as s:
        rows = (
            await s.execute(
                select(ApiKey).where(ApiKey.user_id == request.user.identity)
            )
        ).scalars().all()
    return [_read(r) for r in rows]
```

```ts
export type ApiKey = components["schemas"]["ApiKeyRead"];
export type ApiKeyCreated = components["schemas"]["ApiKeyCreated"];

export async function fetchApiKeys(): Promise<ApiKey[]> {
  const { data } = await api.GET("/api-keys");
  return data ?? [];
}
export async function createApiKey(body: {
  scopes: string[];
  expires_at?: string | null;
}): Promise<ApiKeyCreated> {
  const { data, error } = await api.POST("/api-keys", { body });
  if (error) throw new Error("Failed to create API key");
  return data as ApiKeyCreated;
}
export async function revokeApiKey(id: string): Promise<void> {
  await api.DELETE("/api-keys/{kid}", { params: { path: { kid: id } } });
}
```
`queries.ts` (import + hooks):
```ts
export function useApiKeys() {
  return useQuery({ queryKey: ["api-keys"], queryFn: fetchApiKeys });
}
export function useCreateApiKey() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: createApiKey,
    onSuccess: () => qc.invalidateQueries({ queryKey: ["api-keys"] }),
  });
}
export function useRevokeApiKey() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: revokeApiKey,
    onSuccess: () => qc.invalidateQueries({ queryKey: ["api-keys"] }),
  });
}
```

- [ ] **Step 4: Implement `ApiKeysPage`**

```tsx
import { Button, Code, Modal, MultiSelect, Select, Table, TextInput } from "@mantine/core";
import { notifications } from "@mantine/notifications";
import { useState } from "react";

import type { ApiKey } from "../api/endpoints";
import { useApiKeys, useCreateApiKey, useRevokeApiKey } from "../api/queries";
import { useAuthStore } from "../store/auth";

const EXPIRY_OPTIONS = [
  { value: "never", label: "Never" },
  { value: "7", label: "7 days" },
  { value: "30", label: "30 days" },
  { value: "90", label: "90 days" },
];

function status(k: ApiKey): string {
  if (k.revoked_at) return "revoked";
  if (k.expires_at && new Date(k.expires_at) <= new Date()) return "expired";
  return "active";
}

export function ApiKeysPage() {
  const user = useAuthStore((s) => s.user);
  const { data: keys } = useApiKeys();
  const create = useCreateApiKey();
  const revoke = useRevokeApiKey();
  const [scopes, setScopes] = useState<string[]>([]);
  const [expiry, setExpiry] = useState<string | null>("never");
  const [created, setCreated] = useState<string | null>(null);

  async function onCreate() {
    try {
      const days = expiry === "never" ? null : Number(expiry);
      const expires_at = days
        ? new Date(Date.now() + days * 86_400_000).toISOString()
        : null;
      const res = await create.mutateAsync({ scopes, expires_at });
      setCreated(res.key);
      setScopes([]);
    } catch {
      notifications.show({ message: "Failed to create API key", color: "red" });
    }
  }

  return (
    <div>
      <h1 className="title">API keys</h1>
      <div className="box">
        <MultiSelect
          label="Scopes"
          data={[...new Set(user?.permissions ?? [])]}
          value={scopes}
          onChange={setScopes}
        />
        <Select label="Expires" data={EXPIRY_OPTIONS} value={expiry} onChange={setExpiry} />
        <Button mt="md" onClick={onCreate} loading={create.isPending}>
          Create key
        </Button>
      </div>
      <Table>
        <Table.Thead>
          <Table.Tr>
            <Table.Th>ID</Table.Th>
            <Table.Th>Scopes</Table.Th>
            <Table.Th>Expires</Table.Th>
            <Table.Th>Status</Table.Th>
            <Table.Th>Actions</Table.Th>
          </Table.Tr>
        </Table.Thead>
        <Table.Tbody>
          {(keys ?? []).map((k) => (
            <Table.Tr key={k.id}>
              <Table.Td>
                <Code>{k.id.slice(0, 8)}</Code>
              </Table.Td>
              <Table.Td>{k.scopes.join(", ")}</Table.Td>
              <Table.Td>{k.expires_at ? new Date(k.expires_at).toLocaleString() : "never"}</Table.Td>
              <Table.Td>{status(k)}</Table.Td>
              <Table.Td>
                {status(k) === "active" && (
                  <Button size="xs" color="red" onClick={() => revoke.mutateAsync(k.id)}>
                    Revoke
                  </Button>
                )}
              </Table.Td>
            </Table.Tr>
          ))}
        </Table.Tbody>
      </Table>
      <Modal opened={!!created} onClose={() => setCreated(null)} title="Copy your API key">
        <TextInput
          data-testid="created-key"
          value={created ?? ""}
          readOnly
          onClick={(e) => e.currentTarget.select()}
        />
        <Button
          mt="md"
          onClick={() => created && navigator.clipboard.writeText(created)}
        >
          Copy
        </Button>
      </Modal>
    </div>
  );
}
```

## Notes

- `request.auth.scopes` for a JWT caller equals their role-derived permissions; for an API-key caller it equals that key's scopes. Validating against it satisfies "subset of the user's permissions" for the normal (JWT) case.
- Mantine MultiSelect/Select need the wrapper click in Playwright (same pattern as the RBAC E2E); use `.mantine-MultiSelect-input`/`.mantine-Select-input` if `getByRole("combobox")` is intercepted.
- If expiry/revoked comparisons hit naive/aware datetimes, normalize `expires_at` to UTC (done in `auth_backend` and `_normalize_expiry`).
