# Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Convert the repo into a Uv Workspaces monorepo, define the full SQLModel data model, and implement async SQLite persistence (including an async refactor of the harness session layer so it can be SQLite-backed directly).

**Architecture:** A new `agentbox-models` package holds SQLModel tables and async repositories over `aiosqlite`, with Alembic migrations. It depends on `agentbox-harness` for domain types (`AgentSpec`, `Turn`, `MessageLogStore`) and implements the harness's async `MessageLogStore` protocol as `SQLiteMessageLogStore`.

**Tech Stack:** Python 3.11+, `sqlmodel`, `sqlalchemy[asyncio]`, `aiosqlite`, `alembic`, `pydantic-ai` (harness), `uv` workspaces, `pytest` + `pytest-asyncio`.

## Global Constraints

- Python `>=3.11`.
- **Async everywhere** — all persistence and session-layer operations are `async` (`aiosqlite` + SQLAlchemy `AsyncSession`).
- Use SQLModel for tables; SQLite for storage.
- String primary keys (UUID hex) generated by the application (aligns with the harness's `uuid.uuid4().hex` ids); no auto-increment int PKs.
- **No business logic** — RBAC enforcement, auth, and API-key validation are out of scope; only schema + persistence.
- Alembic for migrations.
- Follow TDD: write the failing test, run it, implement, run it, commit.
- Do not break the existing harness test suite; after the async refactor all harness tests must still pass.

**Reference docs (verify exact signatures while implementing):**
- SQLModel / SQLAlchemy async: `https://sqlmodel.tiangolo.com/`, `https://docs.sqlalchemy.org/en/20/orm/extensions/asyncio.html`.
- Alembic: `https://alembic.sqlalchemy.org/`.
- Uv workspaces: `https://docs.astral.sh/uv/concepts/workspaces/`.

```python
# session.py
from dataclasses import dataclass
from typing import Protocol

from agentbox_harness.spec import AgentSpec


@dataclass(frozen=True)
class Turn:
    id: str
    role: str
    content: str
    parent_id: str | None


class MessageLogStore(Protocol):
    async def append(self, turn: Turn) -> None: ...

    async def list_turns(self) -> list[Turn]: ...

    async def get(self, turn_id: str) -> Turn: ...


class InMemoryMessageLogStore:
    def __init__(self) -> None:
        self._turns: dict[str, Turn] = {}

    async def append(self, turn: Turn) -> None:
        self._turns[turn.id] = turn

    async def list_turns(self) -> list[Turn]:
        return list(self._turns.values())

    async def get(self, turn_id: str) -> Turn:
        return self._turns[turn_id]


@dataclass
class Session:
    session_id: str
    project_root: str
    agent_spec: AgentSpec | None
    store: MessageLogStore
    head_id: str | None

    async def append(self, turn: Turn, parent_id: str | None) -> None:
        await self.store.append(turn)
        self.head_id = turn.id

    async def history(self) -> list[Turn]:
        if self.head_id is None:
            return []
        chain: list[Turn] = []
        current = await self.store.get(self.head_id)
        while True:
            chain.append(current)
            if current.parent_id is None:
                break
            current = await self.store.get(current.parent_id)
        return list(reversed(chain))
```

```python
        self,
        session: Session,
```

```python
# db.py
from sqlalchemy.ext.asyncio import AsyncEngine, async_sessionmaker, create_async_engine
from sqlmodel import SQLModel


def create_engine_and_sessionmaker(database_url: str):
    engine = create_async_engine(database_url)
    factory = async_sessionmaker(engine, expire_on_commit=False)
    return engine, factory


async def init_db(engine: AsyncEngine) -> None:
    async with engine.begin() as conn:
        await conn.run_sync(SQLModel.metadata.create_all)
```

```python
# alembic/env.py
from alembic import context
from sqlalchemy import engine_from_config, pool
from logging.config import fileConfig
from sqlmodel import SQLModel
from agentbox_models.tables.auth import User, Role, Permission, RolePermissionLink, UserRoleLink, ApiKey  # noqa
from agentbox_models.tables.core import Provider, Model, Agent, AgentCard, Project, SessionRow, TurnRow  # noqa

config = context.config
if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = SQLModel.metadata


def run_migrations_offline():
    context.configure(url=config.get_main_option("sqlalchemy.url"), target_metadata=target_metadata, literal_binds=True)
    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online():
    connectable = engine_from_config(config.get_section(config.config_ini_section), prefix="sqlalchemy.", poolclass=pool.NullPool)
    with connectable.connect() as connection:
        context.configure(connection=connection, target_metadata=target_metadata)
        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
```

```python
# repos/agents.py
import json
import uuid

from sqlmodel import select

from agentbox_harness.spec import (
    AgentSpec,
    McpToolBinding,
    Mode,
    ModelBinding,
    NativeToolBinding,
)
from agentbox_models.tables.core import Agent, Model, Provider


def _parse_bindings(raw: str):
    bindings = []
    for item in json.loads(raw or "[]"):
        if "server_url" in item:
            bindings.append(McpToolBinding(**item))
        else:
            bindings.append(NativeToolBinding(**item))
    return bindings


def _parse_modes(raw: str) -> list[Mode]:
    return [Mode(**item) for item in json.loads(raw or "[]")]


class AgentRepository:
    def __init__(self, session_factory):
        self._factory = session_factory

    async def save(self, spec: AgentSpec) -> AgentSpec:
        async with self._factory() as s:
            model = (await s.exec(select(Model).where(Model.name == spec.model_binding.model))).first()
            if model is None:
                raise ValueError(f"Unknown model: {spec.model_binding.model}")
            existing = (await s.exec(select(Agent).where(Agent.name == spec.name))).first()
            row = existing or Agent(id=uuid.uuid4().hex, name=spec.name, description="", instructions="", model_id=model.id)
            row.description = spec.description
            row.instructions = spec.instructions
            row.model_id = model.id
            row.tool_bindings = json.dumps([b.__dict__ for b in spec.tool_bindings])
            row.mode_definitions = json.dumps([{"name": m.name, "flags": m.flags} for m in spec.mode_definitions])
            row.sub_agent_ids = json.dumps(spec.sub_agent_ids)
            s.add(row)
            await s.commit()
        return spec

    async def load(self, name: str) -> AgentSpec | None:
        async with self._factory() as s:
            row = (await s.exec(select(Agent).where(Agent.name == name))).first()
            if row is None:
                return None
            model = (await s.exec(select(Model).where(Model.id == row.model_id))).first()
            provider = (await s.exec(select(Provider).where(Provider.id == model.provider_id))).first()
            return AgentSpec(
                name=row.name,
                description=row.description,
                instructions=row.instructions,
                tool_bindings=_parse_bindings(row.tool_bindings),
                mode_definitions=_parse_modes(row.mode_definitions),
                sub_agent_ids=json.loads(row.sub_agent_ids or "[]"),
                model_binding=ModelBinding(model=model.name, provider=provider.name),
            )

    async def delete(self, name: str) -> None:
        async with self._factory() as s:
            row = (await s.exec(select(Agent).where(Agent.name == name))).first()
            if row is not None:
                await s.delete(row)
                await s.commit()
```

```python
# repos/message_log.py
from sqlmodel import select

from agentbox_harness.session import Turn
from agentbox_models.tables.core import TurnRow


class SQLiteMessageLogStore:
    def __init__(self, session, session_id: str):
        self._session = session
        self._session_id = session_id

    async def append(self, turn: Turn) -> None:
        self._session.add(
            TurnRow(
                id=turn.id,
                session_id=self._session_id,
                role=turn.role,
                content=turn.content,
                parent_id=turn.parent_id,
            )
        )
        await self._session.commit()

    async def list_turns(self) -> list[Turn]:
        rows = (
            await self._session.exec(
                select(TurnRow).where(TurnRow.session_id == self._session_id)
            )
        ).all()
        return [Turn(id=r.id, role=r.role, content=r.content, parent_id=r.parent_id) for r in rows]

    async def get(self, turn_id: str) -> Turn:
        row = (
            await self._session.exec(
                select(TurnRow).where(TurnRow.id == turn_id)
            )
        ).first()
        if row is None:
            raise KeyError(turn_id)
        return Turn(id=row.id, role=row.role, content=row.content, parent_id=row.parent_id)
```

```python
# repos/crud.py
from typing import TypeVar, Generic, Type
from sqlmodel import SQLModel, select

T = TypeVar("T", bound=SQLModel)


class Repository(Generic[T]):
    model: Type[T]

    def __init__(self, session_factory):
        self._factory = session_factory

    async def create(self, obj: T) -> T:
        async with self._factory() as s:
            s.add(obj)
            await s.commit()
            await s.refresh(obj)
            return obj

    async def get(self, id: str) -> T | None:
        async with self._factory() as s:
            return (await s.exec(select(self.model).where(self.model.id == id))).first()

    async def list(self) -> list[T]:
        async with self._factory() as s:
            return list((await s.exec(select(self.model))).all())

    async def delete(self, id: str) -> None:
        async with self._factory() as s:
            row = (await s.exec(select(self.model).where(self.model.id == id))).first()
            if row is not None:
                await s.delete(row)
                await s.commit()

    model = Project
```
