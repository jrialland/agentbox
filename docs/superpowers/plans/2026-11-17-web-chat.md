# Chat (Core + Edit/Fork + Composer) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the agentic chat in `apps/web` — AG-UI streaming, Markdown/Mermaid/reasoning/tool-call rendering, history/resume, mode switching, edit/fork, and a composer with multimodal attachments and speech-to-text — with backend support and Playwright E2E.

**Architecture:** Backend gains a session-attachments table + upload endpoint, an active-branch history endpoint, and send extensions (`mode`, `fork_from_turn_id`, `attachment_ids`) feeding a multimodal prompt; a `test` provider yields deterministic output. Frontend uses `@ag-ui/client` (HttpAgent with a body-rewriting `fetch`) to stream, renders with `react-markdown`/`mermaid`, and adds the composer + edit/fork.

**Tech Stack:** FastAPI/SQLModel/Alembic, pytest; React/TS, `@ag-ui/client`, `@ag-ui/core`, `react-markdown`, `remark-gfm`, `mermaid`, `react-zoom-pan-pinch`, TanStack Query, Playwright.

## Global Constraints

- Backend: `ruff`/`ty`/`bandit` clean; per-package `uv run pytest` green.
- Frontend: `bun run build` green; **functional validation via Playwright E2E**.
- Regenerate the OpenAPI schema after backend changes.
- Attachments are multimodal (`BinaryContent`); the turn log stores text only.

```python
class MessageRead(BaseModel):
    id: str
    role: str
    content: str
    parent_id: str | None = None
    created_at: str


@router.get("/{sid}/messages", response_model=list[MessageRead])
async def list_messages(sid: str, request: Request, _=Depends(require_permission("sessions:read"))):
    async with request.app.state.session_factory() as s:
        rows = (
            await s.execute(select(TurnRow).where(TurnRow.session_id == sid))
        ).scalars().all()
    by_id = {r.id: r for r in rows}
    parent_ids = {r.parent_id for r in rows if r.parent_id}
    leaves = [r for r in rows if r.id not in parent_ids]
    if not leaves:
        return []
    leaf = max(leaves, key=lambda r: (r.created_at or ""))
    chain = []
    cur = leaf
    while cur is not None:
        chain.append(cur)
        cur = by_id.get(cur.parent_id) if cur.parent_id else None
    chain.reverse()
    return [
        MessageRead(
            id=r.id,
            role=r.role,
            content=r.content,
            parent_id=r.parent_id,
            created_at=r.created_at.isoformat() if r.created_at else "",
        )
        for r in chain
    ]
```

```ts
// chat/agent.ts
import { HttpAgent } from "@ag-ui/client";

export interface SendOptions {
  mode: string;
  forkFromTurnId?: string | null;
  attachmentIds?: string[];
}

export function createChatAgent(opts: {
  baseUrl: string;
  token: string | null;
  sessionId: string;
  send: SendOptions;
}) {
  return new HttpAgent({
    url: `${opts.baseUrl}/sessions/${opts.sessionId}/messages`,
    headers: opts.token ? { Authorization: `Bearer ${opts.token}` } : {},
    fetch: (url, init) => {
      const body = JSON.parse((init?.body as string) ?? "{}");
      const messages = body.messages ?? [];
      const lastUser = [...messages].reverse().find((m: { role: string }) => m.role === "user");
      const payload = {
        content: lastUser?.content ?? "",
        mode: opts.send.mode,
        fork_from_turn_id: opts.send.forkFromTurnId ?? null,
        attachment_ids: opts.send.attachmentIds ?? [],
      };
      return fetch(url, { ...init, body: JSON.stringify(payload) });
    },
  });
}
```

// chat/useChat.ts
import { useRef, useState } from "react";
import { createChatAgent, type SendOptions } from "./agent";

export function useChat(baseUrl: string, token: string | null, sessionId: string) {
  const [text, setText] = useState("");
  const [reasoning, setReasoning] = useState("");
  const [toolCalls, setToolCalls] = useState<{ id: string; name: string; args: string; result?: string }[]>([]);
  const [running, setRunning] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function send(content: string, opts: SendOptions) {
    setText(""); setReasoning(""); setToolCalls([]); setError(null); setRunning(true);
    const agent = createChatAgent({ baseUrl, token, sessionId, send: opts });
    agent.addMessage({ id: crypto.randomUUID(), role: "user", content });
    try {
      await agent.runAgent(undefined, {
        onTextMessageContent: ({ textMessageBuffer }: { textMessageBuffer: string }) => setText(textMessageBuffer),
        onReasoningMessageContent: ({ reasoningMessageBuffer }: { reasoningMessageBuffer: string }) => setReasoning(reasoningMessageBuffer),
        onRunErrorEvent: ({ event }: { event: { message?: string } }) => setError(event.message ?? "run error"),
      });
    } finally {
      setRunning(false);
    }
  }
  return { text, reasoning, toolCalls, running, error, send };
}
```

export function speechAvailable(): boolean {
  return typeof window !== "undefined" && Boolean((window as any).SpeechRecognition || (window as any).webkitSpeechRecognition);
}

export function startSpeech(onText: (t: string) => void): () => void {
  const Ctor = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
  if (!Ctor) return () => {};
  const rec = new Ctor();
  rec.lang = "en-US";
  rec.interimResults = true;
  rec.onresult = (e: any) => {
    let text = "";
    for (let i = e.resultIndex; i < e.results.length; i++) text += e.results[i][0].transcript;
    onText(text);
  };
  rec.start();
  return () => rec.stop();
}
```
