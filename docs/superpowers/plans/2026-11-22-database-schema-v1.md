# Database Schema v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the SQLite-flavored, JSON/blob-heavy schema with a normalized, FK-enforced schema v1 that keeps every current feature, using SQLite for this phase.

**Architecture:** Normalize agent configuration, project roots, API-key scopes, turn activity, and turn attachments into relational tables. Move avatar, attachment, and A2A-card bytes into a `media_file`-backed media directory. Delete the historical Alembic chain and create one initial migration. HTTP response shapes stay stable.

**Tech Stack:** Python 3.11, SQLModel/SQLAlchemy 2 async, aiosqlite, Alembic, FastAPI, pytest, React/Vite (type regeneration only).

## Global Constraints

- No `JSON` or `LargeBinary` columns anywhere in `SQLModel.metadata`.
- Every relationship is a real foreign key. Session-owned rows cascade; agent/model/provider/project references restrict.
- SQLite connections enable `PRAGMA foreign_keys=ON`.
- Long free-form text uses `sa.Text`; bounded fields use explicit `sa.String(n)`.
- Human-readable text, including instructions, message content, tool args/results, and reasoning text, must not be truncated by MySQL’s `VARCHAR(255)` default.
- No compatibility layer, alias, or dual write path. Remove old columns, old migrations, and `SQLiteMessageLogStore` naming.
- Keep existing HTTP paths and response fields. `extra_roots`, `tool_bindings`, `mode_definitions`, `scopes`, `attachments`, and `activity` keep their current shapes.
- Media paths are relative, POSIX-style, and generated by the application.
- Do not remove tests to make the suite pass; update their fixtures.
- Backend commands run from repo root. Frontend commands run from `apps/web`.
- Ruff line length is 100; target Python 3.11.
- Every task ends with a commit.

```python
from __future__ import annotations

import hashlib
import os
import tempfile
import uuid
from dataclasses import dataclass
from pathlib import Path

from loguru import logger


@dataclass(frozen=True)
class StoredFile:
    storage_path: str
    media_type: str
    size_bytes: int
    sha256: str


class MediaStore:
    def __init__(self, root: str | Path):
        self.root = Path(root).expanduser().resolve()
        self.root.mkdir(parents=True, exist_ok=True)

    def _validate_category(self, category: str) -> str:
        cleaned = category.strip("/")
        if not cleaned or ".." in Path(cleaned).parts:
            raise ValueError("invalid media category")
        return cleaned

    def _validate_suffix(self, suffix: str) -> str:
        if not suffix:
            return ""
        if not suffix.startswith(".") or "/" in suffix or "\\" in suffix or ".." in suffix:
            raise ValueError("invalid media suffix")
        return suffix

    def _safe_path(self, storage_path: str) -> Path:
        rel = Path(storage_path)
        if rel.is_absolute() or ".." in rel.parts:
            raise ValueError("invalid media path")
        target = (self.root / rel).resolve()
        if target != self.root and self.root not in target.parents:
            raise ValueError("invalid media path")
        return target

    def save(
        self,
        data: bytes,
        *,
        category: str,
        media_type: str,
        suffix: str = "",
    ) -> StoredFile:
        category = self._validate_category(category)
        suffix = self._validate_suffix(suffix)
        rel = f"{category}/{uuid.uuid4().hex}{suffix}"
        target = self._safe_path(rel)
        target.parent.mkdir(parents=True, exist_ok=True)

        fd, tmp_name = tempfile.mkstemp(dir=target.parent, prefix=".tmp-")
        try:
            with os.fdopen(fd, "wb") as handle:
                handle.write(data)
                handle.flush()
                os.fsync(handle.fileno())
            os.replace(tmp_name, target)
        except Exception:
            Path(tmp_name).unlink(missing_ok=True)
            raise

        return StoredFile(
            storage_path=rel,
            media_type=media_type,
            size_bytes=len(data),
            sha256=hashlib.sha256(data).hexdigest(),
        )

    def path(self, storage_path: str) -> Path:
        return self._safe_path(storage_path)

    def read(self, storage_path: str) -> bytes:
        return self.path(storage_path).read_bytes()

    def exists(self, storage_path: str) -> bool:
        return self.path(storage_path).is_file()

    def delete(self, storage_path: str) -> None:
        try:
            self.path(storage_path).unlink()
        except FileNotFoundError:
            logger.warning("media file missing path={}", storage_path)
```

from sqlalchemy import event
from sqlalchemy.ext.asyncio import AsyncEngine, async_sessionmaker, create_async_engine
from sqlmodel import SQLModel


def create_engine_and_sessionmaker(database_url: str, echo: bool = False):
    engine = create_async_engine(database_url, echo=echo)
    if database_url.startswith("sqlite"):
        @event.listens_for(engine.sync_engine, "connect")
        def _enable_sqlite_foreign_keys(dbapi_connection, _record):
            cursor = dbapi_connection.cursor()
            cursor.execute("PRAGMA foreign_keys=ON")
            cursor.close()

    factory = async_sessionmaker(engine, expire_on_commit=False)
    return engine, factory


async def init_db(engine: AsyncEngine) -> None:
    async with engine.begin() as conn:
        await conn.run_sync(SQLModel.metadata.create_all)

async def _load_bindings(session, agent_id: str):
    rows = (
        await session.execute(
            select(AgentToolBinding)
            .where(AgentToolBinding.agent_id == agent_id)
            .order_by(AgentToolBinding.position)
        )
    ).scalars().all()
    bindings = []
    for row in rows:
        if row.kind == "native":
            flags = (
                await session.execute(
                    select(AgentToolBindingRequiredFlag.flag).where(
                        AgentToolBindingRequiredFlag.binding_id == row.id
                    )
                )
            ).scalars().all()
            bindings.append(
                NativeToolBinding(name=row.name or "", required_flags=tuple(flags))
            )
            continue

        args = (
            await session.execute(
                select(AgentToolBindingArg.value)
                .where(AgentToolBindingArg.binding_id == row.id)
                .order_by(AgentToolBindingArg.position)
            )
        ).scalars().all()
        env_rows = (
            await session.execute(
                select(AgentToolBindingEnv).where(
                    AgentToolBindingEnv.binding_id == row.id
                )
            )
        ).scalars().all()
        passthrough = (
            await session.execute(
                select(AgentToolBindingEnvPassthrough.name)
                .where(AgentToolBindingEnvPassthrough.binding_id == row.id)
                .order_by(AgentToolBindingEnvPassthrough.position)
            )
        ).scalars().all()
        bindings.append(
            McpToolBinding(
                server_url=row.server_url or "",
                tool_prefix=row.tool_prefix or "",
                id=row.server_id or "",
                command=row.command or "",
                args=list(args),
                env={item.key: item.value for item in env_rows},
                env_passthrough=list(passthrough),
                cwd=row.cwd or "",
            )
        )
    return bindings
