# Agent Skills Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: `superpowers:executing-plans` (or `subagent-driven-development`). Steps use checkbox (`- [ ]`) syntax.

**Goal:** Discover/activate `SKILL.md` skills with progressive disclosure, a catalog + `activate_skill` tool, `/` explicit activation, a user-level Skills page, and built-in skills.

**Architecture:** `skills-ref` parses/validates skills from built-in/user/project dirs; a catalog is appended to the agent instructions and an `activate_skill` native tool is registered; skill dirs are read-only allowlisted in `FilesystemToolbox`. The chat/ACP/A2A builders use a shared `prepare_agent` helper.

**Tech Stack:** `skills-ref` 0.1, pydantic-ai, SQLModel, React 19, Mantine v9.

## Global Constraints

- Use `skills-ref` for parsing/validation/prompt XML.
- Precedence: project > user > built-in.
- Management UI edits user-level skills only; guard with `agents:create`.
- Skill dirs are read-only; writes stay confined to the project root.
- `ruff`/`ty`/`bandit` clean; existing tests keep passing.

```python
def __init__(self, root_dir, extra_read_roots: list[pathlib.Path] | None = None):
    self.root_dir = pathlib.Path(root_dir).resolve()
    self._read_roots = [self.root_dir, *[pathlib.Path(r).resolve() for r in (extra_read_roots or [])]]

def _resolve(self, path: str, *, write: bool = False) -> pathlib.Path:
    p = (self.root_dir / path).resolve() if not pathlib.Path(path).is_absolute() else pathlib.Path(path).resolve()
    roots = [self.root_dir] if write else self._read_roots
    if not any(p == r or r in p.parents for r in roots):
        raise PathEscapesRootError(path)
    return p
```
Use `_resolve(path, write=True)` in `write_file`, `append_file`, `edit_file`, `create_directory`, `move_file`, `copy_file`; reads use the default. In `search_files`, compute relative against the matched root (or return absolute paths).

```python
import pathlib
from dataclasses import dataclass

from skills_ref import read_properties, to_prompt, validate

from agentbox_harness.spec import AgentSpec, NativeToolBinding


@dataclass
class Skill:
    name: str
    description: str
    location: pathlib.Path
    scope: str


_BUILTIN = pathlib.Path(__file__).parent / "skills_builtin"


def _scan(dir_: pathlib.Path, scope: str) -> list[Skill]:
    out = []
    if not dir_.is_dir():
        return out
    for entry in sorted(dir_.iterdir()):
        skill_md = entry / "SKILL.md"
        if entry.is_dir() and skill_md.is_file():
            try:
                props = read_properties(entry)
            except Exception:
                continue
            if getattr(props, "description", None):
                out.append(Skill(props.name, props.description, skill_md, scope))
    return out


def discover_skills(project_root: str, user_dir: str) -> list[Skill]:
    scopes = {
        "builtin": _scan(_BUILTIN, "builtin"),
        "user": _scan(pathlib.Path(user_dir).expanduser(), "user"),
        "project": _scan(pathlib.Path(project_root) / ".agents" / "skills", "project"),
    }
    merged: dict[str, Skill] = {}
    for scope in ("builtin", "user", "project"):
        for skill in scopes[scope]:
            merged[skill.name] = skill
    return list(merged.values())


def catalog_instructions(skills: list[Skill]) -> str:
    if not skills:
        return ""
    xml = to_prompt([s.location.parent for s in skills])
    return (
        "The following skills provide specialized instructions. "
        "When a task matches a skill's description, call activate_skill with its name.\n"
        + xml
    )


def augment_spec(spec: AgentSpec, skills: list[Skill]) -> AgentSpec:
    if not skills:
        return spec
    return spec.model_copy(
        update={
            "instructions": spec.instructions + "\n\n" + catalog_instructions(skills),
            "tool_bindings": [
                *spec.tool_bindings,
                NativeToolBinding(name="activate_skill"),
            ],
        }
    )


def make_activate_tool(skills: list[Skill]):
    by_name = {s.name: s for s in skills}

    async def activate_skill(name: str) -> str:
        skill = by_name.get(name)
        if skill is None:
            raise ValueError(f"Unknown skill: {name}. Available: {', '.join(by_name)}")
        text = skill.location.read_text(encoding="utf-8")
        body = text.split("---", 2)[-1].strip() if text.startswith("---") else text
        base = skill.location.parent
        try:
            resources = sorted(
                p.relative_to(base).as_posix()
                for p in base.rglob("*")
                if p.is_file() and p.name != "SKILL.md"
            )
        except Exception:
            resources = []
        listing = "\n".join(resources)
        return f"<skill_content name={name!r}>\n{body}\n\nSkill directory: {base}\nResources:\n{listing}\n</skill_content>"

    return activate_skill
```

```python
from agentbox_api.runtime import build_registry, resolve_agent_runtime

@dataclass
class PreparedAgent:
    runtime: object
    spec: AgentSpec
    registry: object
    skills: list[Skill]

async def prepare_agent(factory, agent_name: str, project_root: str, user_dir: str) -> PreparedAgent:
    runtime = await resolve_agent_runtime(factory, agent_name)
    skills = discover_skills(project_root, user_dir)
    spec = augment_spec(runtime.spec, skills)
    extra_tools = {"activate_skill": make_activate_tool(skills)} if skills else None
    registry = build_registry(
        project_root,
        extra_read_roots=[s.location.parent for s in skills],
        extra_tools=extra_tools,
    )
    return PreparedAgent(runtime, spec, registry, skills)
```
