# Backend (FastAPI) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `agentbox-api` — a FastAPI service exposing the platform's use cases (agents, providers, projects, sessions, chat streaming) with Starlette-native JWT/API-key auth and RBAC, wiring `agentbox-models`, `agentbox-providers`, and `agentbox-harness`.

**Architecture:** A FastAPI app with a `pydantic-settings` config, a Starlette `AuthenticationBackend` (JWT + API key) that returns `(AuthCredentials, User)`, an `AuthenticationMiddleware`, `Depends`-based `require_permission`, routers per resource, and a chat endpoint that runs the harness and streams AG-UI SSE via pydantic-ai's `AGUIAdapter`.

**Tech Stack:** Python 3.11+, FastAPI, uvicorn, Starlette, pydantic-ai (`[ag-ui]`), pydantic-settings, pyjwt, bcrypt, SQLModel/aiosqlite (via `agentbox-models`), httpx (tests), pytest + pytest-asyncio.

## Global Constraints

- Python `>=3.11`.
- **Use framework extension points, don't hand-roll**: Starlette `AuthenticationBackend`/`AuthCredentials`/`BaseUser`/`AuthenticationMiddleware`, `pydantic-settings.BaseSettings`, `fastapi.security.HTTPBearer`, pydantic-ai `AGUIAdapter`, SQLModel models as request/response bodies where the wire format matches.
- String UUID primary keys (aligned with `agentbox-models`).
- Async everywhere (async endpoints, `AsyncSession`, `httpx.AsyncClient`+`ASGITransport` in tests).
- No real model calls in tests — use `TestModel`/`FunctionModel`; the real Ollama path is covered in `agentbox-providers`.
- Follow TDD; run `ruff`/`ty`/`bandit` and `pytest` before each commit.

**Reference docs (verify exact signatures while implementing):**
- Starlette auth: `https://www.starlette.io/authentication/`.
- pydantic-ai AG-UI: `https://ai.pydantic.dev/integrations/ui/ag-ui/`.

```python
# security.py
import hashlib
from datetime import datetime, timedelta, timezone

import bcrypt
import jwt


def hash_password(password: str) -> str:
    return bcrypt.hashpw(password.encode(), bcrypt.gensalt()).decode()


def verify_password(password: str, hashed: str) -> bool:
    return bcrypt.checkpw(password.encode(), hashed.encode())


def create_jwt(user_id: str, secret: str, algorithm: str, expiry_seconds: int) -> str:
    payload = {
        "sub": user_id,
        "exp": datetime.now(timezone.utc) + timedelta(seconds=expiry_seconds),
    }
    return jwt.encode(payload, secret, algorithm=algorithm)


def decode_jwt(token: str, secret: str, algorithm: str) -> dict:
    return jwt.decode(token, secret, algorithms=[algorithm])


def hash_api_key(raw: str) -> str:
    return hashlib.sha256(raw.encode()).hexdigest()
```

```python
# auth_backend.py
from starlette.authentication import AuthCredentials, AuthenticationBackend, BaseUser

from sqlmodel import select

from agentbox_models.tables.auth import (
    ApiKey,
    Permission,
    Role,
    RolePermissionLink,
    User,
    UserRoleLink,
)
from agentbox_api.config import Settings
from agentbox_api.security import decode_jwt, hash_api_key


class CurrentUser(BaseUser):
    def __init__(self, id: str, username: str, permissions: set[str]):
        self.id = id
        self.username = username
        self.permissions = permissions

    @property
    def is_authenticated(self) -> bool:
        return True

    @property
    def display_name(self) -> str:
        return self.username

    @property
    def identity(self) -> str:
        return self.id


async def load_user_permissions(session, user_id: str) -> set[str]:
    role_ids = (
        await session.execute(select(UserRoleLink.role_id).where(UserRoleLink.user_id == user_id))
    ).scalars().all()
    if not role_ids:
        return set()
    perm_ids = (
        await session.execute(
            select(RolePermissionLink.permission_id).where(
                RolePermissionLink.role_id.in_(role_ids)
            )
        )
    ).scalars().all()
    if not perm_ids:
        return set()
    perms = (await session.execute(select(Permission.name).where(Permission.id.in_(perm_ids)))).scalars().all()
    return set(perms)


class JWTAuthenticationBackend(AuthenticationBackend):
    def __init__(self, session_factory, settings: Settings):
        self._factory = session_factory
        self._settings = settings

    async def authenticate(self, request):
        auth = request.headers.get("Authorization")
        if not auth or not auth.startswith("Bearer "):
            return None
        token = auth.removeprefix("Bearer ")

        async with self._factory() as s:
            # Try JWT
            try:
                payload = decode_jwt(
                    token, self._settings.jwt_secret.get_secret_value(), self._settings.jwt_algorithm
                )
                user_id = payload["sub"]
                user = (await s.execute(select(User).where(User.id == user_id))).scalars().first()
                if user is None:
                    return None
                perms = await load_user_permissions(s, user_id)
                return AuthCredentials(scopes=list(perms)), CurrentUser(user.id, user.username, perms)
            except Exception:
                pass

            # Try API key
            key_row = (
                await s.execute(select(ApiKey).where(ApiKey.key_hash == hash_api_key(token)))
            ).scalars().first()
            if key_row is None:
                return None
            user = (await s.execute(select(User).where(User.id == key_row.user_id))).scalars().first()
            if user is None:
                return None
            import json
            scopes = json.loads(key_row.scopes or "[]")
            return AuthCredentials(scopes=scopes), CurrentUser(user.id, user.username, set(scopes))
```

```python
from fastapi import HTTPException, Request
from sqlalchemy.ext.asyncio import AsyncSession


async def get_session(request: Request) -> AsyncSession:
    factory = request.app.state.session_factory
    async with factory() as session:
        yield session

    async def checker(request: Request):
        scopes = request.auth.scopes if request.auth is not None else []
        if name not in scopes:
            raise HTTPException(status_code=403, detail="Forbidden")
        return request.user
    return checker
```

```python
# app.py
from contextlib import asynccontextmanager

from fastapi import FastAPI
from starlette.middleware.authentication import AuthenticationMiddleware

from agentbox_api.auth_backend import JWTAuthenticationBackend
from agentbox_api.config import Settings
from agentbox_models.db import create_engine_and_sessionmaker


def create_app(settings: Settings) -> FastAPI:
    @asynccontextmanager
    async def lifespan(app: FastAPI):
        engine, factory = create_engine_and_sessionmaker(settings.database_url)
        app.state.engine = engine
        app.state.session_factory = factory
        yield
        await engine.dispose()

    app = FastAPI(lifespan=lifespan)
    app.add_middleware(
        AuthenticationMiddleware,
        backend=JWTAuthenticationBackend(app.state.session_factory, settings),
    )
    return app
```

Note: `app.state.session_factory` is set during lifespan, but the middleware backend needs it at construction. Store the settings on the app and resolve the factory lazily inside the backend, or set `app.state` before middleware registration. Verify and adjust: simplest is to create the engine/factory *before* building the app, pass the factory to the backend, and dispose the engine in the lifespan.

```python
# streaming.py
from pydantic_ai.ui.ag_ui import AGUIAdapter

from agentbox_harness.events import RunEvent
from agentbox_harness.errors import RunOutcome


async def _unwrap(run_events):
    async for e in run_events:
        if isinstance(e.event, RunOutcome):
            # Surface error/cancel as a terminal AG-UI event (RUN_ERROR / RUN_FINISHED(cancelled)).
            continue  # handled by the adapter's run-complete/error path via the outcome
        yield e.event


def agui_response(run_events):
    adapter = AGUIAdapter()
    stream = adapter.build_event_stream()
    agui = stream.transform_stream(_unwrap(run_events))
    return stream.streaming_response(agui)
```

```python
from agentbox_api.streaming import agui_response

from agentbox_models.tables.core import Model, Provider, Project, SessionRow


class MessageRequest(BaseModel):
    content: str


@router.post("/{sid}/messages")
async def chat(sid: str, body: MessageRequest, request: Request, _=Depends(require_permission("sessions:create"))):
    factory = request.app.state.session_factory
    async with factory() as s:
        srow = (await s.execute(select(SessionRow).where(SessionRow.id == sid))).scalars().first()
        if srow is None:
            raise HTTPException(status_code=404, detail="Not found")
        agent = (await s.execute(select(AgentRow).where(AgentRow.id == srow.agent_id))).scalars().first()
        model = (await s.execute(select(Model).where(Model.id == agent.model_id))).scalars().first()
        provider = (await s.execute(select(Provider).where(Provider.id == model.provider_id))).scalars().first()
        project = (await s.execute(select(Project).where(Project.id == srow.project_id))).scalars().first()

    spec = await AgentRepository(factory).load(agent.name)
    resolved = resolve_model(provider, model.name)

    root = pathlib.Path(project.path)
    box = FilesystemToolbox(root)
    registry = ToolRegistry()
    registry.register_native("read_text_file", box.read_text_file, required_flags=("read",))
    registry.register_native("write_file", box.write_file, required_flags=("write",))
    registry.register_native("list_directory", box.list_directory, required_flags=("read",))

    store = SQLiteMessageLogStore(s, sid)
    session = Session(session_id=sid, project_root=project.path, agent_spec=spec, store=store, head_id=None)
    session.mode_name = "agent"

    harness = Harness(registry=registry, model=resolved, store_factory=lambda: store)
    bus = harness._run_and_subscribe(session, body.content)  # see note below

    return agui_response(bus)
```
